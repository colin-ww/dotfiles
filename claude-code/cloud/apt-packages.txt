# System packages for R-setup.sh (one or more per line, "#" starts a comment).
# Checked in CI (check.sh): every name must exist in apt on Ubuntu noble.
# WATCH OUT: R pkg says "cannot open shared object libXYZ.so" -> a system lib
#   is missing. Add its -dev package here (the -dev pulls in the runtime lib).

# R itself (CRAN's repo, newer than Ubuntu's). r-base-dev = compilers/headers
# so a pkg with no binary can still build from source.
r-base r-base-dev

# curl/ssl -> httr, curl     xml2 -> xml2, rvest     uv -> fs
libcurl4-openssl-dev libssl-dev
libxml2-dev
libuv1-dev

# ragg, textshaping (ggplot2 graphics)
libfontconfig1-dev
libharfbuzz-dev
libfribidi-dev
libfreetype-dev     # NOT libfreetype6-dev: virtual pkg on noble
libpng-dev libtiff-dev libjpeg-dev   # libtiff5-dev was jammy, noble is libtiff-dev

# igraph (-> targets, crew). Depends on libglpk40 (libglpk.so.40), so no need
# to list it. NOT "libglpk40-dev": doesn't exist, broke setup on 2026-11-13.
libglpk-dev

# odbc (libodbc.so.2; sc-nts-frame box::use()s odbc)
unixodbc-dev

# dev tools for editing the repos (check.sh uses both; apt versions are fine)
codespell shellcheck
